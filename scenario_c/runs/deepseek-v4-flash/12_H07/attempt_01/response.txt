import json
import gurobipy as gp
from gurobipy import GRB

suppliers = ["A", "B", "C", "D", "E", "F"]
components = ["C1", "C2", "C3"]
demand = {"C1": 80, "C2": 70, "C3": 60}
fixed_cost = {"A": 120, "B": 140, "C": 100, "D": 150, "E": 90, "F": 110}
capacity = {"A": 100, "B": 110, "C": 90, "D": 120, "E": 80, "F": 95}
risk = {"A": 2, "B": 3, "C": 1, "D": 4, "E": 2, "F": 3}

unit_cost = {
    "A": {"C1": 9, "C2": 12, "C3": 11},
    "B": {"C1": 10, "C2": 9, "C3": 12},
    "C": {"C1": 11, "C2": 10, "C3": 9},
    "D": {"C2": 11, "C3": 10},
    "E": {"C1": 12, "C2": 8},
    "F": {"C1": 10, "C3": 8},
}

quality = {
    "A": {"C1": 96, "C2": 94, "C3": 95},
    "B": {"C1": 93, "C2": 97, "C3": 92},
    "C": {"C1": 98, "C2": 95, "C3": 96},
    "D": {"C2": 94, "C3": 93},
    "E": {"C1": 97, "C2": 98},
    "F": {"C1": 94, "C3": 98},
}

available = [(i, j) for i in suppliers for j in components if j in unit_cost[i]]

m = gp.Model("sourcing")
m.setParam("OutputFlag", 0)

x = {}
z = {}

for i in suppliers:
    for j in components:
        if (i, j) in available:
            x[i, j] = m.addVar(lb=0, vtype=GRB.INTEGER, name=f"x_{i}_{j}")
            z[i, j] = m.addVar(vtype=GRB.BINARY, name=f"z_{i}_{j}")

y = {i: m.addVar(vtype=GRB.BINARY, name=f"y_{i}") for i in suppliers}

m.addConstrs((x[i, j] <= capacity[i] * z[i, j] for (i, j) in available), name="pair_upper")
m.addConstrs((x[i, j] >= 10 * z[i, j] for (i, j) in available), name="pair_lower")
m.addConstrs((z[i, j] <= y[i] for (i, j) in available), name="activation")

m.addConstrs(
    (sum(x[i, j] for i in suppliers if (i, j) in available) == demand[j] for j in components),
    name="demand"
)
m.addConstrs(
    (sum(x[i, j] for j in components if (i, j) in available) <= capacity[i] for i in suppliers),
    name="capacity"
)

m.addConstr(sum(y[i] for i in suppliers) <= 4, name="max_suppliers")
m.addConstr(y["A"] + y["B"] >= 1, name="min_AB")

m.addConstrs(
    (sum(z[i, j] for i in suppliers if (i, j) in available) >= 2 for j in components),
    name="min_two_suppliers"
)

m.addConstr(sum(risk[i] * x[i, j] for (i, j) in available) <= 580, name="risk")

for j in components:
    m.addConstr(
        sum(quality[i][j] * x[i, j] for i in suppliers if (i, j) in available) >= 95 * demand[j],
        name=f"quality_{j}"
    )

m.setObjective(
    sum(unit_cost[i][j] * x[i, j] for (i, j) in available)
    + sum(fixed_cost[i] * y[i] for i in suppliers),
    GRB.MINIMIZE
)

m.optimize()

if m.status == GRB.OPTIMAL:
    variables = {v.VarName: float(v.X) for v in m.getVars()}
    result = {"status": "OPTIMAL", "objective": float(m.ObjVal), "variables": variables}
else:
    result = {"status": str(m.status), "objective": 0.0, "variables": {}}

print("SOLUTION_JSON=" + json.dumps(result, separators=(",", ":")))
